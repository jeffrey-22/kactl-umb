\section{Dynamic programming}
	\subsection{Quadrangle Inequality}
		Quadrangle Inequality: cost function $C$ satisfies for all $l < r$: $C(l,r+1)-C(l, r) \ge C(l+1,r+1) - C(l+1,r)$.
		Alternatively $C(a,c) + C(b,d) \le C(a,d) + C(b,c)$ for all $a \le b \le c \le d$.
		Define $\text{opt}(r)=$(leftmost or rightmost)$l(< r)$ that minimize $C(l,r)$, then $\text{opt}$ is non-decreasing.
		Useful for decision points of DP transitions. Consider writing assertions of monotonicity of $\text{opt}$.

	\subsection{Sum over Subset}

		Keep set $S(\text{mask}, i)$ as sum of (subset of first $i$ bits of $\text{mask}$ + remaining bits of $\text{mask}$).
		Loop over $i$ first, then $\text{mask}$. If $\text{mask}$'s $i^{th}$ bit is $1$, 
		additionally include $S(\text{mask} \oplus 2^i, i-1)$.
		Get rid of the second dimension for memory optimization.

	\subsection{Slope trick}

		For 2D DP $f_i(x)$ that transitions from $f_{i-1}(y)$ that is convex, consider the slopes $g_i(x)=f_i(x+1)-f_i(x)$, they must be non-decreasing. Maintain slope changing points with some data structure. (Example: maintain set $v$ such that $v_{t}=\min_{g_i(p)=t} p$.)

	\subsection{Subset sum}

		To find if a subset sum of weights can be equal to $k$, group items into form $(w_i, occ_i)$, and decompose into powers of $2$. So $(w_i,12)=\{w_i,2w_i,4w_i,5w_i\}$. This is $O(\frac{C^{1.5}}{32})$.

	\subsection{$O(nk)$ tree DP}

		If design DP as $dp[u][i+j] \leftarrow f(dp[v][i], dp[w][j])$ for all pairs of children $(v, w)$, with length of $dp[u] \le k$, this is $O(nk)$.

	\subsection{Alien's Trick}
	
		Let the max score of using sth $x$ times be $f(x)$, and we need to find $f(K)$ for a fixed $K$. If $f(x)$ is concave or convex, then $f(x) - \lambda x$ will increase, stay the same, and decrease. Let $v(\lambda)$ be max of $f(x)-\lambda x$ (find by DP with $\lambda$ as penalty), and $c(\lambda)$ the rightmost $x$ that maximizes $f(x)-\lambda x$. Binary search for highest $\lambda_0$ such that $c(\lambda_0) \ge K$, then answer is $v(\lambda_0) + \lambda_0 K$.

\section{Debugging tricks}
	\begin{itemize}
		\item \verb@signal(SIGSEGV, [](int) { _Exit(0); });@ converts segfaults into Wrong Answers.
			Similarly one can catch SIGABRT (assertion failures) and SIGFPE (zero divisions).
			\verb@_GLIBCXX_DEBUG@ failures generate SIGABRT (or SIGSEGV on gcc 5.4.0 apparently).
		\item Generator with seed \verb@main(int argc, char* argv[])@ and \verb@srand(atoi(argv[1]))@
		\item Stress bash \verb@for((i=1;;++i)); do ... done@ and \verb@./gen $i > r diff -w w1 w2 || break@
	\end{itemize}

\section{Optimization tricks}
	\verb@__builtin_ia32_ldmxcsr(40896);@ disables denormals (which make floats 20x slower near their minimum value). Needs 64-bit OS.
	\subsection{Bit hacks}
		\begin{itemize}
			\item \verb@x & -x@ is the least bit in \texttt{x}.
			\item \verb@for (int x = m; x; ) { --x &= m; ... }@ loops over all subset masks of \texttt{m} (except \texttt{m} itself).
			\item \verb@c = x&-x, r = x+c; (((r^x) >> 2)/c) | r@ is the next number after \texttt{x} with the same number of bits set.
			\item \verb@rep(b,0,K) rep(i,0,(1 << K))@ \\ \verb@  if (i & 1 << b) D[i] += D[i^(1 << b)];@ computes all sums of subsets.
		\end{itemize}
	\subsection{Pragmas}
		\begin{itemize}
			\item \lstinline{#pragma GCC optimize ("Ofast")} will make GCC auto-vectorize loops and optimizes floating points better.
			\item \lstinline{#pragma GCC target ("avx2")} can double performance of vectorized code, but causes crashes on old machines. Consider putting loops over arrays in a separate method.
			\item \lstinline{#pragma GCC optimize ("trapv")} kills the program on integer overflows (but is really slow).
		\end{itemize}
